declare(strict_types=1);

namespace WooPulse\Core;

final class MetricsEngine {
    private const CACHE_TTL = 300; // 5 mins Redis transient

    public function get_realtime_velocity(): array {
        $transient_key = 'woopulse_live_velocity';
        $cached = wp_cache_get($transient_key, 'woopulse');

        if (false !== $cached) {
            return $cached;
        }

        // Direct indexed query bypassing postmeta bloat
        $metrics = $this->query_hpos_orders_fast();
        wp_cache_set($transient_key, $metrics, 'woopulse', self::CACHE_TTL);

        return $metrics;
    }
}
